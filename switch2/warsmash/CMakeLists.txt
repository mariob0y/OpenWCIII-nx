# Warsmash on the Switch 2: the C++ that Clearwing VM makes of Warsmash + gdx + SwitchGDX's Java side.
#   bash warsmash/build-java.sh                      compile + transpile into warsmash/build/gen/src
#   (devkitPro MSYS2)
#   cmake -S warsmash -B build-warsmash -G "Unix Makefiles" -DCMAKE_BUILD_TYPE=Release -DCMAKE_TOOLCHAIN_FILE=/opt/devkitpro/libounce/cmake/Switch2.cmake
#   make -C build-warsmash -j16
#   (not Ninja: with response files it writes the flags, MSYS paths included, into them, which the
#   native compiler cannot read)
# A unity build: build-java.sh batches the ~7,000 transpiled files into ~500 translation units
# (warsmash/unity.py, build/unity), which keeps the number of compiler processes down.
#
#   python scripts/check_switch2_safety.py build-warsmash/warsmash.elf   before it goes onto a console
#
# W3: the executable. Natives: backend/native (our Switch 2 replacement of SwitchGDX's SwitchGDX.cpp,
# generated by scripts/make_nx_switchgdx.py; the audio mixer on audout; the GL loader) and, as
# they are, SwitchGDX's FreeType, gdx2d, ETC1 and Platform sources. backend/java's inline natives
# come out of the transpiler with the rest (build/gen/src).
cmake_minimum_required(VERSION 3.13)
project(warsmash_nx C CXX)
# Thousands of objects: the link line is longer than Windows allows. The Makefile generator can pass
# just the objects in a response file (relative paths: MSYS does not translate /c/... paths inside
# one for the Windows-native compiler, so everything else stays on the command line).
set(CMAKE_CXX_USE_RESPONSE_FILE_FOR_OBJECTS 1)
set(CMAKE_CXX_USE_RESPONSE_FILE_FOR_LIBRARIES 0)

if(NOT NINTENDO_SWITCH2)
    message(FATAL_ERROR "This is for the Switch 2 toolchain only (libounce/cmake/Switch2.cmake)")
endif()

set(GEN "${CMAKE_CURRENT_SOURCE_DIR}/build/gen/src")
if(NOT EXISTS "${GEN}/main.cpp")
    message(FATAL_ERROR "No transpiled sources: run warsmash/build-java.sh first")
endif()

set(TP "${CMAKE_CURRENT_SOURCE_DIR}/../third_party")
set(ZZIP_ROOT "${TP}/pkg-zziplib/opt/devkitpro/portlibs/switch")

set(SGDX_RES "${CMAKE_CURRENT_SOURCE_DIR}/../external/switch-gdx/switch-gdx/res")
set(NATIVE "${CMAKE_CURRENT_SOURCE_DIR}/backend/native")
set(SWITCH1_PORTLIBS "${DEVKITPRO}/portlibs/switch")

# The batches of build/gen/src that warsmash/unity.py writes (each #includes its files from ${GEN}).
file(GLOB GEN_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/build/unity/*.cpp")
if(NOT GEN_SOURCES)
    message(FATAL_ERROR "No unity batches: run warsmash/build-java.sh (or python -I warsmash/unity.py batches build/gen/src build/unity)")
endif()
# The transpiled code: an object library as in W2, its flags unchanged: flags.make is a dependency
# of every object, so any change there (even a per-file include directory, which CMake writes into
# it) recompiles all ~5,800 objects, 85 minutes. backend/java's inline natives find
# backend/native's headers in build/gen/src, where build-java.sh copies them.
add_library(warsmash_gen OBJECT ${GEN_SOURCES})
set_target_properties(warsmash_gen PROPERTIES CXX_STANDARD 20 CXX_STANDARD_REQUIRED ON)
target_include_directories(warsmash_gen PRIVATE "${GEN}" "${TP}/prefix-switch2/include" "${ZZIP_ROOT}/include"
    "${DEVKITPRO}/portlibs/switch/include")
# As probe B: no JIT on the console (java.lang.reflect.Proxy unavailable); Java's wrapping int
# arithmetic (-fwrapv) and unfused a*b+c (-ffp-contract=off).
target_compile_definitions(warsmash_gen PRIVATE ASMJIT_STATIC ASMJIT_NO_JIT)
target_compile_options(warsmash_gen PRIVATE -O2 -fwrapv -ffp-contract=off -Wno-volatile -Wno-return-type -Wno-attributes)
add_executable(warsmash $<TARGET_OBJECTS:warsmash_gen>
    "${NATIVE}/nx_switchgdx.cpp" "${NATIVE}/nx_audio.cpp" "${NATIVE}/nxgl.c" "${NATIVE}/nx_threads.cpp"
    "${SGDX_RES}/switchgdx/FreeType_native.cpp" "${SGDX_RES}/switchgdx/gdx2d.c"
    "${SGDX_RES}/switchgdx/etc1_utils.cpp" "${SGDX_RES}/switchgdx/Platform.cpp")
set_target_properties(warsmash PROPERTIES CXX_STANDARD 20 CXX_STANDARD_REQUIRED ON)
target_include_directories(warsmash PRIVATE "${GEN}" "${NATIVE}" "${SGDX_RES}" "${SGDX_RES}/switchgdx"
    "${DEVKITPRO}/portlibs/switch/include" "${SWITCH1_PORTLIBS}/include/freetype2")
target_compile_options(warsmash PRIVATE -O2 -fwrapv -ffp-contract=off -Wno-volatile -Wno-return-type -Wno-attributes)
# EGL as probe A (the Switch 2 portlibs); FreeType (+ HarfBuzz, PNG, bzip2, zlib) and mpg123 from
# the Switch 1 portlibs (plain AArch64 code; probe B linked their zlib); FreeType and HarfBuzz need
# each other, hence the group.
target_link_libraries(warsmash PRIVATE
    "${DEVKITPRO}/portlibs/switch2/lib/libEGL.a"
    "${TP}/prefix-switch2/lib/libffi.a"
    "${ZZIP_ROOT}/lib/libzzip.a"
    "${SWITCH1_PORTLIBS}/lib/libmpg123.a"
    -Wl,--start-group
    "${SWITCH1_PORTLIBS}/lib/libfreetype.a" "${SWITCH1_PORTLIBS}/lib/libharfbuzz.a"
    "${SWITCH1_PORTLIBS}/lib/libpng16.a" "${SWITCH1_PORTLIBS}/lib/libbz2.a" "${SWITCH1_PORTLIBS}/lib/libz.a"
    -Wl,--end-group)
# Default-core threads (the GL driver's) start on a core of their own, not the game thread's
# (backend/native/nx_threads.cpp's __wrap_threadCreate, as Torchlight's platform_profiler.cpp).
target_link_options(warsmash PRIVATE -Wl,--wrap=threadCreate -Wl,--require-defined=__wrap_threadCreate)

# The guard: no service, mount or SD-card write outside /switch/warsmash*/; and the crash handler.
include(${CMAKE_CURRENT_LIST_DIR}/../cmake/switch2_guard.cmake)
ws_switch2_guard(warsmash ADD_SOURCE)

nx_generate_nacp(warsmash.nacp NAME "Warsmash" AUTHOR "warsmash-nx" VERSION "0.1")
nx_create_nro(warsmash NACP warsmash.nacp ICON ${CMAKE_CURRENT_SOURCE_DIR}/icon.jpg)  # scripts/make_icon.py
